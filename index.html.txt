/* ── [2번] Wilder's RSI (EMA 방식) ── */
function calcRSI(prices, n = 14) {
  if (prices.length < n + 1) return 50;
  let gains = [], losses = [];
  for (let i = 1; i < prices.length; i++) {
    const d = prices[i] - prices[i-1];
    gains.push(d > 0 ? d : 0); losses.push(d < 0 ? -d : 0);
  }
  let avgG = gains.slice(0, n).reduce((a, b) => a + b) / n;
  let avgL = losses.slice(0, n).reduce((a, b) => a + b) / n;
  for (let i = n; i < gains.length; i++) {
    avgG = (avgG * (n - 1) + gains[i]) / n;
    avgL = (avgL * (n - 1) + losses[i]) / n;
  }
  return avgL === 0 ? 100 : parseFloat((100 - (100 / (1 + avgG/avgL))).toFixed(1));
}

/* ── [1번] Notion 데이터 전송 ── */
async function sendToNotion(tradeData) {
  const key = localStorage.getItem('notion_key'), dbId = localStorage.getItem('notion_db_id');
  if (!key || !dbId) return;
  const proxy = "https://corsproxy.io/?", url = "https://api.notion.com/v1/pages";
  try {
    await fetch(proxy + encodeURIComponent(url), {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json', 'Notion-Version': '2022-06-28' },
      body: JSON.stringify({
        parent: { database_id: dbId },
        properties: {
          "종목명": { title: [{ text: { content: tradeData.name } }] },
          "구분": { select: { name: tradeData.action } },
          "가격": { number: tradeData.price },
          "수익률": { rich_text: [{ text: { content: tradeData.pnl || "-" } }] },
          "기록시간": { date: { start: new Date().toISOString() } }
        }
      })
    });
    toast('Notion 동기화 완료', 'info');
  } catch (e) { console.error(e); }
}

function saveSettings() {
  localStorage.setItem('chipbot_key', document.getElementById('apiKeyInput').value);
  localStorage.setItem('notion_key', document.getElementById('notionKey').value);
  localStorage.setItem('notion_db_id', document.getElementById('notionDbId').value);
  toast('설정이 저장되었습니다.', 'info');
}

function toast(m, t='info') {
  const w = document.getElementById('toastwrap'), d = document.createElement('div');
  d.className = `toast show t-${t}`; d.textContent = m;
  w.appendChild(d); setTimeout(() => { d.remove(); }, 3000);
}

function updateUI() {
  const prices = pH[cur], c = prices[prices.length-1];
  const rsi = calcRSI(prices);
  document.getElementById('hPrice').textContent = `₩${c.toLocaleString()}`;
  document.getElementById('rsiV').textContent = rsi;
  document.getElementById('rsiV').style.color = rsi < 30 ? 'var(--green)' : rsi > 70 ? 'var(--red)' : 'var(--txt)';
  
  if (!pChart) {
    const ctx = document.getElementById('pc').getContext('2d');
    pChart = new Chart(ctx, {
      type: 'line',
      data: { labels: prices.map((_,i)=>i), datasets: [{ data: prices, borderColor: '#f87171', borderWidth: 2, pointRadius: 0, tension: 0.2 }] },
      options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { x: { display: false }, y: { grid: { color: 'rgba(255,255,255,0.05)' } } } }
    });
  } else {
    pChart.data.datasets[0].data = prices; pChart.update();
  }
}

async function runAI() {
  const rsi = calcRSI(pH[cur]);
  const action = rsi < 35 ? "매수 권고" : rsi > 65 ? "매도 권고" : "관망";
  document.getElementById('aiRep').textContent = `[Wilder's RSI: ${rsi}] 현재 지표상 ${action} 상태입니다.`;
  
  // 분석 시 노션에 상태 기록 (옵션)
  sendToNotion({ name: STOCKS[cur].name, action: "AI 분석", price: pH[cur].at(-1), pnl: `RSI: ${rsi}` });
}

  cur = k; if(pChart) { pChart.destroy(); pChart = null; }
  updateUI();
}

}
